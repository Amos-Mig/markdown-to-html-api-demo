What you deploy


# mdapi.py — the whole service is one file
from flask import Flask, request, jsonify
import renderer          # bundled, stdlib-only modules

app = Flask(__name__)

@app.post('/md')
def md():
    html = renderer.render(request.get_data(), sanitize=True)
    return jsonify(html=html)
